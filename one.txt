import os
from dataclasses import dataclass

@dataclass
class Config:
    """Configuration for the churn prediction system"""
    
    # GCP Settings
    project_id: str = os.getenv('GCP_PROJECT_ID', 'your-project-id')
    location: str = os.getenv('GCP_LOCATION', 'us-central1')
    gcs_bucket: str = os.getenv('GCS_BUCKET', 'churn-feature-store-demo')
    
    # Feature Store Settings
    featurestore_id: str = os.getenv('FEATURESTORE_ID', 'churn_featurestore')
    entity_type_id: str = os.getenv('ENTITY_TYPE_ID', 'customer')
    
    # Data Settings
    num_customers: int = int(os.getenv('NUM_CUSTOMERS', '5000'))
    
    # Model Settings
    model_path: str = os.getenv('MODEL_PATH', 'models/churn_model.pkl')
