from pyspark.sql import SparkSession
from pyspark.sql.functions import (
    rand, randn, when, col, expr, monotonically_increasing_id,
    round as spark_round, floor, lit
)
from pyspark.sql.types import *
import random

from orange.utils import generate_bank_data


def start_data():
    data_path = "./bank_loan_data"

    # Initialize Spark
    spark = SparkSession.builder \
        .appName("udf") \
        .config("spark.sql.shuffle.partitions", "200") \
        .config("spark.sql.execution.arrow.pyspark.enabled", "true") \
        .config("spark.driver.memory", "4g") \
        .config("spark.executor.memory", "4g") \
        .getOrCreate()

    print("data generated started")

    generate_bank_data(
        spark,
        num_records=1_000_000,  # Generate 1M records (increase for production)
        output_path=data_path
    )
    print("data generated finished")

if __name__=="__main__":
    start_data()
