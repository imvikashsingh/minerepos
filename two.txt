# model_training.py

import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from sklearn.linear_model import LogisticRegression
import joblib
from pyspark.sql import SparkSession
from pyspark.sql.functions import col

# Initialize Spark Session for reading Parquet and data prep
spark = SparkSession.builder.appName("BankLoanDefaultTraining").getOrCreate()

# Read loans training data and customers data from Parquet
loans_train_df = spark.read.parquet("loans_train.parquet")
customers_df = spark.read.parquet("customers.parquet")

# Data Preparation: Join loans with customers on customer_id
joined_df = loans_train_df.join(customers_df, on="customer_id", how="inner")

# Add a new derived feature (credit_income_ratio)
joined_df = joined_df.withColumn("credit_income_ratio", col("credit_score") / col("income"))

# Select relevant columns and convert to Pandas for scikit-learn
feature_columns = ["age", "income", "loan_amount", "credit_score", "years_employed", "num_dependents", "credit_income_ratio"]
train_pd = joined_df.select(*feature_columns, "default").toPandas()

# Stop Spark Session
spark.stop()

# Separate features and target
X_train = train_pd[feature_columns]
y_train = train_pd["default"]

# Define scikit-learn Pipeline
pipeline = Pipeline([
    ("scaler", StandardScaler()),  # Scale features
    ("classifier", LogisticRegression(random_state=42))  # Logistic Regression
])

# Train the model
pipeline.fit(X_train, y_train)

# Save the trained model
joblib.dump(pipeline, "loan_default_model.pkl")
