# prediction.py

from pyspark.sql import SparkSession
from pyspark.sql.functions import pandas_udf, PandasUDFType
from pyspark.sql.types import DoubleType
import pandas as pd
import joblib
from pyspark.ml.evaluation import BinaryClassificationEvaluator, MulticlassClassificationEvaluator
from pyspark.sql.functions import col

# Initialize Spark Session
spark = SparkSession.builder.appName("BankLoanDefaultPrediction").getOrCreate()

# Load the trained scikit-learn model and broadcast it
model = joblib.load("loan_default_model.pkl")
broadcast_model = spark.sparkContext.broadcast(model)

# Define pandas_udf for distributed predictions
feature_columns = ["age", "income", "loan_amount", "credit_score", "years_employed", "num_dependents", "credit_income_ratio"]

@pandas_udf(DoubleType(), PandasUDFType.SCALAR)
def predict_udf(*cols) -> pd.Series:
    # Reconstruct DataFrame from columns
    df = pd.DataFrame({feature: col for feature, col in zip(feature_columns, cols)})
    # Access the broadcasted model
    clf = broadcast_model.value
    # Predict using the scikit-learn pipeline
    return pd.Series(clf.predict(df).astype(float))

# Read loans test data and customers data from Parquet
loans_test_df = spark.read.parquet("loans_test.parquet")
customers_df = spark.read.parquet("customers.parquet")

# Data Preparation: Join loans with customers on customer_id
joined_df = loans_test_df.join(customers_df, on="customer_id", how="inner")

# Add a new derived feature (credit_income_ratio)
joined_df = joined_df.withColumn("credit_income_ratio", col("credit_score") / col("income"))

# Make predictions using pandas_udf (distributed)
predictions = joined_df.withColumn("prediction", predict_udf(*feature_columns))

# Select relevant columns for viewing
predictions.select("default", "prediction", "customer_id").show(5)

# Evaluate the model
# Accuracy
evaluator_accuracy = MulticlassClassificationEvaluator(labelCol="default", predictionCol="prediction", metricName="accuracy")
accuracy = evaluator_accuracy.evaluate(predictions)

# AUC for binary classification
evaluator_auc = BinaryClassificationEvaluator(labelCol="default", rawPredictionCol="prediction", metricName="areaUnderROC")
auc = evaluator_auc.evaluate(predictions)

# Precision, Recall, F1 (weighted)
evaluator_f1 = MulticlassClassificationEvaluator(labelCol="default", predictionCol="prediction", metricName="f1")
f1 = evaluator_f1.evaluate(predictions)

evaluator_precision = MulticlassClassificationEvaluator(labelCol="default", predictionCol="prediction", metricName="weightedPrecision")
precision = evaluator_precision.evaluate(predictions)

evaluator_recall = MulticlassClassificationEvaluator(labelCol="default", predictionCol="prediction", metricName="weightedRecall")
recall = evaluator_recall.evaluate(predictions)

print(f"Accuracy: {accuracy:.2f}")
print(f"AUC: {auc:.2f}")
print(f"F1 Score: {f1:.2f}")
print(f"Precision: {precision:.2f}")
print(f"Recall: {recall:.2f}")

# Stop Spark Session
spark.stop()
