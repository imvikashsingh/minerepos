# prediction_without_udf.py
# This version uses toPandas() which collects all joined data to the driver node (single node processing),
# potentially leading to out-of-memory issues for very large datasets.

import pandas as pd
from sklearn.metrics import accuracy_score, classification_report
import joblib
from pyspark.sql import SparkSession
from pyspark.sql.functions import col
import time
import sys
import gc

# Initialize Spark Session
spark = SparkSession.builder.appName("BankLoanDefaultPredictionWithoutUDF").getOrCreate()

print("=== PREDICTION WITHOUT PANDAS_UDF (Single Node - Driver Only) ===")

# Read loans test data and customers data from Parquet
loans_test_df = spark.read.parquet("loans_test.parquet")
customers_df = spark.read.parquet("customers.parquet")

print(f"Loans test partitions: {loans_test_df.rdd.getNumPartitions()}")
print(f"Customers partitions: {customers_df.rdd.getNumPartitions()}")

# Data Preparation: Join loans with customers on customer_id
print("Step 1: Performing JOIN (distributed)...")
start_join = time.time()
joined_df = loans_test_df.join(customers_df, on="customer_id", how="inner")
join_duration = time.time() - start_join
print(f"Join completed in {join_duration:.2f} seconds")

# Add a new derived feature (credit_income_ratio)
joined_df = joined_df.withColumn("credit_income_ratio", col("credit_score") / col("income"))

# Select relevant columns
feature_columns = ["age", "income", "loan_amount", "credit_score", "years_employed", "num_dependents", "credit_income_ratio"]
final_columns = feature_columns + ["default"]
processed_df = joined_df.select(*final_columns)

print(f"Step 2: Converting to Pandas (collects ALL data to DRIVER - MEMORY RISK)...")
start_pandas = time.time()
try:
    test_pd = processed_df.toPandas()  # THIS IS THE CRITICAL STEP THAT CAUSES OOM
    pandas_duration = time.time() - start_pandas
    print(f"toPandas() completed in {pandas_duration:.2f} seconds")
    print(f"Dataset size after toPandas(): {test_pd.shape}")
except MemoryError as e:
    print(f"MEMORY ERROR: {e}")
    print("Out of memory during toPandas() - dataset too large for driver!")
    sys.exit(1)

# Separate features and target
X_test = test_pd[feature_columns]
y_test = test_pd["default"]

print("Step 3: Loading model and making predictions...")
start_predict = time.time()

# Load the trained model
pipeline = joblib.load("loan_default_model.pkl")

# Make predictions
y_pred = pipeline.predict(X_test)

predict_duration = time.time() - start_predict
print(f"Predictions completed in {predict_duration:.2f} seconds")

# Evaluate
accuracy = accuracy_score(y_test, y_pred)
report = classification_report(y_test, y_pred)

print(f"\n=== RESULTS (Without UDF) ===")
print(f"Accuracy: {accuracy:.2f}")
print("Classification Report:\n", report)

# Total duration
total_duration = time.time() - start_join
print(f"\n=== TIMING BREAKDOWN (Without UDF) ===")
print(f"Join: {join_duration:.2f}s")
print(f"toPandas(): {pandas_duration:.2f}s")
print(f"Prediction: {predict_duration:.2f}s")
print(f"TOTAL DURATION: {total_duration:.2f}s")

# Memory cleanup
del test_pd, X_test, y_test, y_pred
gc.collect()

print(f"\nWARNING: This approach collects {len(final_columns)} columns x {processed_df.count()} rows to single driver node!")
print("For datasets > 100K rows, this will likely cause OutOfMemoryError!")

# Stop Spark Session
spark.stop()
