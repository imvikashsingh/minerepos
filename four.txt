import pandas as pd
import numpy as np
from datetime import datetime
from typing import Tuple

class FeatureEngineer:
    """Simple feature engineering for churn prediction"""
    
    @staticmethod
    def engineer_features(customers: pd.DataFrame, 
                         transactions: pd.DataFrame) -> pd.DataFrame:
        """Engineer features from raw data for Feature Store"""
        
        print("🔄 Engineering features...")
        
        # Aggregate transaction features
        tx_features = transactions.groupby('customer_id').agg({
            'amount': ['mean', 'sum', 'count'],
            'transaction_date': ['max'],
            'is_online': ['mean']
        }).reset_index()
        
        # Flatten column names
        tx_features.columns = ['customer_id', 'avg_transaction_amount', 
                              'total_transaction_amount', 'transaction_count',
                              'last_transaction_date', 'online_transaction_ratio']
        
        # Calculate additional features
        current_date = datetime.now()
        tx_features['days_since_last_transaction'] = (
            current_date - pd.to_datetime(tx_features['last_transaction_date'])
        ).dt.days
        
        tx_features['transaction_frequency'] = tx_features['transaction_count'] / (
            tx_features['days_since_last_transaction'] + 1
        ) * 30  # Monthly frequency
        
        # Fill missing values (customers with no transactions)
        tx_features = tx_features.fillna({
            'avg_transaction_amount': 0,
            'total_transaction_amount': 0,
            'transaction_count': 0,
            'days_since_last_transaction': 999,
            'online_transaction_ratio': 0,
            'transaction_frequency': 0
        })
        
        # Merge customer and transaction features
        features = customers.merge(tx_features, on='customer_id', how='left')
        
        # Fill any remaining nulls
        features = features.fillna(0)
        
        print(f"✅ Engineered {len(features)} feature records")
        return features
    
    @staticmethod
    def add_churn_labels(features: pd.DataFrame) -> pd.DataFrame:
        """Add churn labels for training"""
        
        # Simple rule-based churn labeling
        features['churned'] = 0
        
        # High churn risk based on behavior
        conditions = (
            (features['is_active_member'] == 0) |
            (features['transaction_count'] < 3) |
            (features['days_since_last_transaction'] > 60) |
            (features['tenure_months'] < 6)
        )
        
        # Add some randomness to labels
        churn_prob = np.random.uniform(0, 1, len(features))
        features.loc[conditions & (churn_prob > 0.4), 'churned'] = 1
        
        print(f"✅ Added churn labels: {features['churned'].sum()} churned customers")
        return features
