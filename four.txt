from pyspark.sql import SparkSession
from pyspark.sql.functions import pandas_udf, PandasUDFType
from pyspark.sql.types import DoubleType, StructType, StructField
import pandas as pd
import joblib
from pyspark.ml.evaluation import BinaryClassificationEvaluator, MulticlassClassificationEvaluator
from pyspark.sql.functions import col, spark_partition_id
import time
import os

def model_prediction_udf():
    # Initialize Spark Session

    # Measure time for prediction
    start_time = time.time()
    spark = SparkSession.builder.appName("udf").getOrCreate()

    # Load the trained scikit-learn model and broadcast it
    model = joblib.load("loan_model.joblib")
    broadcast_model = spark.sparkContext.broadcast(model)

    # Define pandas_udf for distributed predictions (returns prediction and pid for verification)
    feature_columns = ["age", "income", "loan_amount", "credit_score", "credit_income_ratio"]

    # Use GROUPED_MAP to return multiple columns (prediction and pid)
    schema = StructType([
        StructField("prediction", DoubleType(), False),
        StructField("pid", DoubleType(), False)  # Process ID to verify distribution
    ])

    @pandas_udf(schema, PandasUDFType.GROUPED_MAP)
    def predict_udf(pdf: pd.DataFrame) -> pd.DataFrame:
        clf = broadcast_model.value
        preds = clf.predict(pdf[feature_columns]).astype(float)
        pids = [os.getpid()] * len(pdf)
        return pd.DataFrame({'prediction': preds, 'pid': pids})

    # Read test data from Parquet
    test_df = spark.read.parquet("test_data.parquet")

    # Print number of partitions (if >1 and in cluster mode, UDF processes distributed across executors)
    print(f"Number of partitions in test_df: {test_df.rdd.getNumPartitions()}")


    # Apply predictions using pandas_udf (distributed processing)
    predictions = test_df.groupBy(spark_partition_id()).apply(predict_udf)
    # Note: GROUPED_MAP with spark_partition_id() ensures processing per partition, showing distribution.

    # To verify distribution: Collect unique pids (if multiple, it's distributed across processes/executors)
    unique_pids = predictions.select("pid").distinct().collect()
    print(f"Unique process IDs (if >1, distributed): {[row.pid for row in unique_pids]}")

    # Show sample predictions
    predictions.select("prediction", "pid").show(5)

    duration = time.time() - start_time
    print(f"Total duration for prediction (with UDF): {duration:.2f} seconds")

    spark.stop()


if __name__=="__main__":
    model_prediction_udf()
